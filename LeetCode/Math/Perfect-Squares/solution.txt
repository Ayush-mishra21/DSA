        int ans = 0, res = 0;
        for(int i = 1;  i*i <= n; i++){
            int N = n;
            while(N % i == 0){
                res += Math.pow(i, 2);
                if(res == n){
                    return ans;
                }
                ans++;
                N /= i;
            }
        }
        return res;
