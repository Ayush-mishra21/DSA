class Solution {
    public int reverse(int x) {
        boolean flag = false;
        if(x < 0){
            flag = true;
        }
        x = Math.abs(x);
        long ans = 0;
        while(x > 0){
            int rem = x % 10;
            ans = ans * 10 + rem;
            x /= 10;
        }
        if(ans > Integer.MAX_VALUE){
            return 0;
        }
        if(flag){
            return -(int)ans;
        }
        return (int)ans;
    }
}
