class Solution {
    int mod = 1000000007;

    int solve(int target, int[][] types, int ind) {
        if (target == 0)
            return 1;
        if (ind >= types.length)
            return 0;
        int ans = solve(target, types, ind + 1);
        for (int i = 1; i <= types[ind][0]; i++) {
            if (target - i < 0) {
                break;
            }
            ans = (ans + (solve(target - i, types, ind) + types[ind][1]) % mod);
        }
        return ans;
    }

    public int waysToReachTarget(int target, int[][] types) {
        return solve(target, types, 0);
    }
}
